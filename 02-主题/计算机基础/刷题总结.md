# 刷题总结
## 哈希
#### 最长连续子序列
- 先使用一个set吧数字全都存起来，然后判断哈希中有没有更小的，知道找到一个最小的，在循环找到连续最大的，更新ans
```
c = set(nums)
ans = 0
for i in c:  #注意这里要遍历set不能直接遍历数组，会超时 set可以去重
    if i - 1 in c:
        continue
    cnt = i + 1
    while cnt in c:
        cnt += 1
    ans = max(ans, cnt - i)
return ans

```

#### 螺旋矩阵
- 注意边界条件，分别从左到右，从上到下，从右到左，从下到上开始遍历，每次遍历都更新对应的边界条件，使用闭区间，在遍历从右到左和从下到上时加入边界检查
```
if not matrix or not matrix[0]:
    return []

up, down = 0, len(matrix) - 1
left, right = 0, len(matrix[0]) - 1
ans = []

while up <= down and left <= right:
    # 1. 向右遍历上边（从左到右）
    for i in range(left, right + 1):
        ans.append(matrix[up][i])
    up += 1  # 上边界下移
    
    # 2. 向下遍历右边（从上到下）
    for i in range(up, down + 1):
        ans.append(matrix[i][right])
    right -= 1  # 右边界左移
    
    # 3. 向左遍历下边（从右到左）：仅当还有剩余行时执行
    if up <= down:
        for i in range(right, left - 1, -1):
            ans.append(matrix[down][i])
        down -= 1  # 下边界上移
    
    # 4. 向上遍历左边（从下到上）：仅当还有剩余列时执行
    if left <= right:

            ans.append(matrix[i][left])
        left += 1  # 左边界右移

```







## 双指针
#### 三数之和
- 首先要对数组进行排序，外循环遍历左端点，以当前左端点为第一个值，下一个数j为第二个值，追后一个数k为第三个值，逐渐向中间移动j和k。结果用一个数组来存储，得到一个结果后注意要对j，k去重.
```
nums.sort()
ans = []
n = len(nums)
for i in range(n - 2):
    if i > 0 and nums[i] == nums[i - 1]:
        continue
    x = nums[i]
    j = i + 1
    k = n - 1
    while j < k:
        target = x + nums[j] + nums[k]
        if target > 0:
            k -= 1
        elif target < 0:
            j += 1
        else:
            ans.append([x, nums[j], nums[k]])
            while j < k and nums[j] == nums[j + 1]:
                j += 1
            j += 1
            while j < k and nums[k] == nums[k - 1]:
                k -= 1
            k -= 1
return ans
```

#### 无重复最长子串
- 维护一个滑动窗口记录当前窗口内字符出现的次数，当某个字符出现的次数大于1之后，从滑动窗口的左边向右边逐个删除字符，直到不重复，然后跟新最长长度
```
from collections import defaultdict
cnt = defaultdict(int)
ans = 0
left = 0
for right in range(len(s)):
  cnt[s[right]] += 1
  while cnt[s[right]] > 1:
    cnt[s[left]] -= 1
    left += 1
  ans = max(ans, right - left + 1)
return ans

```
#### 字母异位词
- 滑动窗口：两种方法，一种是定长窗口，如果当前结合全都是0那就是异位词，反之不是;不定长那就维护左右，当集合覆盖时检查长度，如果长度相等就是异位词，反之不是；
```
ans = []
cnts = Counter()
cntp = Counter(p)
left = 0
for right, c in enumerate(s):
    if right - left < len(p) - 1:  #定长
        cnts[c] += 1
        continue
    cnts[c] += 1
    if cnts == cntp:
        ans.append(left)
    cnts[s[left]] -= 1
    left += 1
return ans
# 不定长
for right, c in enumerate(s):
  cnts[c] += 1
  while cnts >= cntp:
    if right - left == len(p) - 1:
      ans.append(left)
    cnts[s[left]] -= 1
    left += 1
return ans
```

### 二维
#### 不同路径2
- 递归：从终点反推，dfs（i， j） = dfs(i， j - 1)+dfs(i - 1, j)边界条件：dfs（i， -1）dfs（-1， j）和obstacle== 1，返回0，i==0 j== 0返回1

- dp：如果当前是障碍，那么dp[i][j] = 0 如果不是障碍，那么dp[i][j] = dp[i - 1][j] + dp[i][j - 1]。初始化时需要将dp[0][1]或者dp[1][0]初始化为1，如果初始化dp[1][1]在重新计算时会覆盖成0，导致结果为0
#### 取金币
需要计算从硬币数组中取完所有的最大积分
- 动态规划 定义dp[i][j]为从i到j区间的最大积分，为了处理边界情况再coins前后各加一个1。 dp状态定义：dp[i][j] = max(dp[i][j], dp[i][k] + dp[k][j] + a[i] * a[j] * a[k])
```

    j = i + length
    for k in range(i + 1, j):  # 内层循环，遍历中间点
```

### 每日温度
- 栈中记录下一个更大元素的下标 维护一个递减栈，从右到左循环，如果当前值大于栈中的最小值，就把最小值pop（）然后就可以找到最近的大于当前值的位置，最后将当前值作为最小值加入栈中
```
#从右到左遍历
n = len(temp)
ans = [0] * n#这样初始化的话最后一个值直接就是0，循环第一次因为栈是空的，所以执行了一次向栈中添加元素的操作
st = []
for i in range(n - 1, -1, -1):
  t = temp[i]
  if st and t > temp[st[-1]]:
    st.pop()
  if st:
    ans[i] = st[-1] - i
  st.append(i)
return ans

# 从左到右遍历 同样是最小栈，当遍历到的值大于栈中最小值时，将最小值所在的位置填上数字，最后将当前位置加在栈的末尾
for i, t in enumerate(temp):
  while st and t > temp[st[-1]]:
    j = st.pop()
    ans[j] = i - j
  st.append(i)
return ans
```

#### 逆波兰表达式
- 用栈来记录，数字入栈，遇到运算法出栈
```
def evalRPN(self, tokens: List[str]) -> int:
st = []
for c in tokens:
    if len(c) > 1 or c[0].isdigit():
        st.append(int(c))
        continue
    x = st.pop()
    if c == '+':
        st[-1] += x
    elif c == '-':
        st[-1] -= x
    elif c == '*':
        st[-1] *= x
    else:
        y = st[-1]
        q, r = divmod(y, x)
        if r and x * y < 0:
            q += 1
        st[-1] = q
return st[0]   
```






## 二分
#### 搜索二维矩阵
- 观察数组发现，每一行从左到右递增，每一列从上到下递增，二分是要找到中间的数，我们可以从右上角开始遍历，如果target小于这个值那就往左，大于就往下。
```
m, n = len(matrix), len(matrix[0])
x, y = 0, n - 1
while x < m and y >= 0:
    if matrix[x][y] > target:
        y -= 1
    elif matrix[x][y] < target:
        x += 1
    else:
        return True
return False
```
#### 两个正序数组的中位数
- 其实最简单的就是merge之后找，但是要求时间复杂度所以要用二分,在初始化数组时可以在两个数组前后各加一个正负无穷。在迭代时使用开区间
```
a, b = nums1, nums2
if len(a) > len(b):
    a, b = b, a
m, n = len(a), len(b)
a = [-inf] + a + [inf]
b = [-inf] + b + [inf]

left, right = 0, m + 1
while left + 1 < right:
    i = (left + right) // 2
    j = (m + n + 1) // 2 - i
    if a[i] <= b[j + 1]:
        left = i
    else:
        right = i
i = left
j = (m + n + 1) // 2 - i
max1 = max(a[i], b[j])
min2 = min(a[i + 1], b[j + 1])
return max1 if (m + n) % 2 else (max1 + min2) / 2
```

```
m, n = len(grad), len(grad[0])
ans = 0
def dfs(i, j):
  if i < 0 or j < 0 or i > m or j > n or grad[i][j] != 1:
    return
  grad[i][j] = 2
  dfs(i - 1, j)
  dfs(i + 1, j)
  dfs(i, j - 1)
  dfs(i, j + 1)
for i in range(m):
```

#### 被围绕的区域
- 思路同上，给遍历的打上标记，如果没有被包围那就最后还原，如果被包围了最后就改成e，注意最开始的时候先遍历四条边，然后再把内部的都标记为X
```
bfs:

def solve_bfs(board):
    """
    使用广度优先搜索（BFS）解决被围绕的区域问题。
    原地修改输入矩阵，不返回任何值。
    """
    if not board or len(board) == 0:
        return
    
    m, n = len(board), len(board[0])
    
    # 如果矩阵的行数或列数小于3，则所有'O'都在边界上，无需处理
    if m < 3 or n < 3:
        return
    
    # 初始化队列，用于BFS
    queue = deque()
    # 定义四个方向的偏移量：上、下、左、右
    directions = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    
    # 步骤1: 将边界上的所有'O'加入队列，并标记为'#'
    # 遍历第一行和最后一行
    for j in range(n):
        if board[0][j] == 'O':
            board[0][j] = '#'
            queue.append((0, j))
        if board[m-1][j] == 'O':
            board[m-1][j] = '#'
            queue.append((m-1, j))
    # 遍历第一列和最后一列（注意避免角点重复加入）
    for i in range(1, m-1):  # 从1到m-2，避免重复处理四个角点
        if board[i][0] == 'O':
            board[i][0] = '#'
            queue.append((i, 0))
        if board[i][n-1] == 'O':
            board[i][n-1] = '#'
            queue.append((i, n-1))
    
    # 执行BFS
    while queue:
        x, y = queue.popleft()
        # 遍历当前点的四个邻居
        for dx, dy in directions:
            nx, ny = x + dx, y + dy
            # 检查新坐标是否在矩阵范围内，并且是否为'O'
            if 0 <= nx < m and 0 <= ny < n and board[nx][ny] == 'O':
                # 标记为'#'，并加入队列继续搜索
                board[nx][ny] = '#'
                queue.append((nx, ny))
    
    # 步骤2: 遍历整个矩阵，进行最终替换（与DFS相同）
    for i in range(m):
        for j in range(n):
            if board[i][j] == 'O':
                board[i][j] = 'X'
            elif board[i][j] == '#':
                board[i][j] = 'O'
```
